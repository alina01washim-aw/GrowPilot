// cppcheck suppression file for MycoControl firmware
//
// Rationale:
// - ArduinoJson headers use advanced token-pasting macros that cppcheck cannot
//   expand, causing false-positive preprocessor errors.
// - Our functions are used across multiple .cpp translation units; cppcheck's
//   single-file analysis reports them as unused.
// - External libraries (Adafruit, OneWire, Sensirion, DallasTemperature) contain
//   non-explicit single-argument constructors and other style warnings that are
//   intentional API design or upstream issues we can't fix.

# ArduinoJson macro expansion false positives (cppcheck cannot handle ARDUINOJSON_CONCAT_)
preprocessorErrorDirective:*preprocessor.hpp

# Cross-TU false positives — functions are used across .cpp files
# Use both forward and backslash patterns for Windows compatibility
unusedFunction:src\*
unusedFunction:src/*
unusedFunction:*\src\*
unusedFunction:*/src/*

# External library style warnings (non-explicit single-argument constructors)
noExplicitConstructor:*Adafruit*
noExplicitConstructor:*OneWire*
noExplicitConstructor:*Sensirion*
noExplicitConstructor:*DallasTemperature*

# External library style warnings (missing override specifier)
missingOverride:*Adafruit*
missingOverride:*OneWire*
missingOverride:*DallasTemperature*

# External library warnings (C++ standard library / framework code - not our code)
# These are in framework/toolchain headers and are not actionable
